#!/usr/bin/env python3
"""Independent P4 verification of WatchLab demo artifacts. Stdlib only."""
import hashlib
import json
import glob
import os
import sys

PROJ = "/home/Komi/projects/watchlab/watchlab-pilot"
FORBIDDEN_PORTS = {2379, 2380}
results = []


def check(name, ok, detail):
    results.append((name, ok, detail))
    print(f"{'PASS' if ok else 'FAIL'} | {name} | {detail}")


def load_jsonl(path):
    recs = []
    with open(path, "r", encoding="utf-8") as f:
        for i, line in enumerate(f, 1):
            line = line.strip()
            if line:
                recs.append(json.loads(line))
    return recs


# --- locate newest demo dir -------------------------------------------------
dirs = sorted(glob.glob("/tmp/watchlab-demo-*"), key=os.path.getmtime)
demo = dirs[-1]
demo_summary_path = os.path.join(demo, "demo-summary.json")
check("demo_dir_found", len(dirs) > 0, f"newest={demo} (all={len(dirs)} dirs, newest mtime={os.path.getmtime(demo):.0f})")

summary = json.load(open(demo_summary_path, encoding="utf-8"))
N = summary.get("events_per_phase")
check("3a_events_per_phase", N == 5, f"events_per_phase={N}")

# --- 3a: demo-summary.json ---------------------------------------------------
overall = summary.get("overall")
check("3a_overall_PASS", overall == "PASS", f"overall={overall}")

scens = summary.get("scenarios", [])
names = [s.get("name") for s in scens]
scen_ok = all(s.get("status") == "PASS" for s in scens) if "status" in scens[0] else all(s.get("ok") for s in scens)
check("3a_scenarios_present_PASS", len(scens) == 2 and scen_ok, f"scenarios={names}, statuses={[s.get('status', s.get('ok')) for s in scens]}")

iso = summary.get("isolation", {})
endpoint = iso.get("endpoint", "")
port = int(endpoint.rsplit(":", 1)[1])
check("3a_port_not_2379_2380", port not in FORBIDDEN_PORTS,
      f"endpoint={endpoint}, client_port={iso.get('client_port')}, peer_port={iso.get('peer_port')}, avoided={iso.get('reserved_ports_avoided')}")
dd = iso.get("data_dir", "")
check("3a_data_dir_under_tmp_not_project", dd.startswith("/tmp/") and not dd.startswith(PROJ),
      f"data_dir={dd}")
demo_id = summary.get("demo_id")
client_port, peer_port = iso.get("client_port"), iso.get("peer_port")
total_verified = summary.get("total_verified_events", summary.get("total_verified_events_count"))
final_ckpt = summary.get("final_checkpoint")
print(f"INFO | demo_id={demo_id} client_port={client_port} peer_port={peer_port} "
      f"events_per_phase={N} total_verified_events={total_verified} final_checkpoint={final_ckpt}")

# --- 3b: reconnect -----------------------------------------------------------
rd = os.path.join(demo, "reconnect")
p1 = load_jsonl(os.path.join(rd, "producer-recon-p1.jsonl"))
p2 = load_jsonl(os.path.join(rd, "producer-recon-p2.jsonl"))
ack_types = set(r["type"] for r in p1) | set(r["type"] for r in p2)
check("3b_producer_p1_p2", len(p1) == N and len(p2) == N and ack_types == {"put_ack"},
      f"p1={len(p1)} p2={len(p2)} records, types={sorted(ack_types)}")

wi = load_jsonl(os.path.join(rd, "watch-initial.jsonl"))
n_wi_wc = sum(1 for r in wi if r["type"] == "watch_created")
n_wi_ev = sum(1 for r in wi if r["type"] == "watch_event")
check("3b_watch_initial_counts", n_wi_wc == 1 and n_wi_ev == N,
      f"watch_created={n_wi_wc}, watch_event={n_wi_ev} (expected 1/{N})")

wr = load_jsonl(os.path.join(rd, "watch-resumed.jsonl"))
wr_ev = [r for r in wr if r["type"] == "watch_event"]
wr_wc = [r for r in wr if r["type"] == "watch_created"]
max_initial = max(r["event_revision"] for r in wi if r["type"] == "watch_event")
check("3b_watch_resumed_first_rev", wr_ev[0]["event_revision"] == max_initial + 1,
      f"first resumed event_revision={wr_ev[0]['event_revision']}, max initial event_revision={max_initial}, expected {max_initial + 1}")
check("3b_watch_resumed_counts", len(wr_ev) == N and len(wr_wc) == 1,
      f"watch_created={len(wr_wc)}, watch_event={len(wr_ev)} (expected 1/{N})")

# cross-check acks vs watch events
ack_map = {}
for r in p1 + p2:
    rid = r["request_id"]
    ack_map.setdefault(rid, []).append(r)
watch_map = {}
for r in list(wi) + list(wr):
    if r["type"] != "watch_event":
        continue
    val = json.loads(r["value"])
    watch_map.setdefault(val["request_id"], []).append((r, val))

matched = missing = dup_w = dup_a = mismatches = 0
missing_ids, dup_ids, mismatch_detail = [], [], []
for rid, acks in ack_map.items():
    if len(acks) > 1:
        dup_a += len(acks) - 1
        dup_ids.append(rid)
    got = watch_map.pop(rid, None)
    if got is None:
        missing += 1
        missing_ids.append(rid)
        continue
    if len(got) > 1:
        dup_w += len(got) - 1
        dup_ids.append(rid)
    r, val = got[0]
    if r["key"] == acks[0]["key"] and r["event_revision"] == acks[0]["revision"] and \
            hashlib.sha256(r["value"].encode("utf-8")).hexdigest() == acks[0]["value_sha256"]:
        matched += 1
    else:
        mismatches += 1
        mismatch_detail.append(f"{rid}: key={r['key']}/{acks[0]['key']} rev={r['event_revision']}/{acks[0]['revision']}")
unexpected = len(watch_map)
total_expected = 2 * N
check("3b_crosscheck", matched == total_expected and missing == 0 and dup_w == 0 and dup_a == 0 and unexpected == 0 and mismatches == 0,
      f"expected={total_expected} matched={matched} missing={missing} duplicates(dup_w={dup_w},dup_a={dup_a}) unexpected={unexpected} mismatches={mismatches} missing_ids={missing_ids} dup_ids={dup_ids} mismatch_detail={mismatch_detail}")

# --- 3c: compaction ----------------------------------------------------------
cd = os.path.join(demo, "compaction")
q = {}
for ph in ("p1", "p2", "p3"):
    q[ph] = load_jsonl(os.path.join(cd, f"producer-comp-{ph}.jsonl"))
check("3c_producer_p1_p2_p3", all(len(q[p]) == N and all(r["type"] == "put_ack" for r in q[p]) for p in ("p1", "p2", "p3")),
      f"p1={len(q['p1'])} p2={len(q['p2'])} p3={len(q['p3'])} (expected {N} each), all put_ack")

ri = load_jsonl(os.path.join(cd, "recover-initial.jsonl"))
ri_snap = [r for r in ri if r["type"] == "snapshot"]
ri_wc = [r for r in ri if r["type"] == "watch_created"]
ri_ev = [r for r in ri if r["type"] == "watch_event"]
check("3c_recover_initial", len(ri_snap) == 1 and len(ri_wc) == 1 and len(ri_ev) == 0,
      f"snapshot={len(ri_snap)} watch_created={len(ri_wc)} watch_event={len(ri_ev)} (expected 1/1/0)")

last_p2_rev = q["p2"][-1]["revision"]
ra = load_jsonl(os.path.join(cd, "recover-after.jsonl"))
ra_wc = [r for r in ra if r["type"] == "watch_created"]
ra_wx = [r for r in ra if r["type"] == "watch_canceled"]
ra_snap = [r for r in ra if r["type"] == "snapshot"]
ra_ev = [r for r in ra if r["type"] == "watch_event"]
last_p1_rev = q["p1"][-1]["revision"]

ok_wc = len(ra_wc) >= 2
ok_wx = len(ra_wx) == 1 and ra_wx[0]["compact_revision"] == last_p2_rev
ok_snap = len(ra_snap) == 1 and ra_snap[0]["snapshot_revision"] == last_p2_rev
ok_wc1 = len(ra_wc) >= 1 and ra_wc[0]["start_revision"] == last_p1_rev + 1
ok_wc2 = len(ra_wc) >= 2 and ra_wc[1]["start_revision"] == last_p2_rev + 1
ok_ev = len(ra_ev) == N
check("3c_recover_after", ok_wc and ok_wx and ok_snap and ok_wc1 and ok_wc2 and ok_ev,
      f"watch_canceled={len(ra_wx)} (compact_revision={ra_wx[0]['compact_revision'] if ra_wx else None}, expected {last_p2_rev}); "
      f"snapshot={len(ra_snap)} (snapshot_revision={ra_snap[0]['snapshot_revision'] if ra_snap else None}, expected {last_p2_rev}); "
      f"watch_created={len(ra_wc)} starts={[r['start_revision'] for r in ra_wc]} (expected [{last_p1_rev + 1},{last_p2_rev + 1},...]); "
      f"watch_event={len(ra_ev)} (expected {N})")

st = json.load(open(os.path.join(cd, "state.json"), encoding="utf-8"))
keys = st["keys"]
all_acks = q["p1"] + q["p2"] + q["p3"]
ack_by_key = {}
for r in all_acks:
    ack_by_key.setdefault(r["key"], []).append(r)

extra = set(keys) - set(ack_by_key)
miss_key = set(ack_by_key) - set(keys)
state_bad = []
for key, acks in ack_by_key.items():
    ack = acks[0]
    if key not in keys:
        continue
    rec = keys[key]
    if rec["mod_revision"] != ack["revision"]:
        state_bad.append(f"{key}: mod_revision={rec['mod_revision']} != ack revision={ack['revision']}")
    if hashlib.sha256(rec["value"].encode("utf-8")).hexdigest() != ack["value_sha256"]:
        state_bad.append(f"{key}: stored value sha256 != ack value_sha256")

state_ok = (st["snapshot_revision"] == last_p2_rev and st["last_revision"] == q["p3"][-1]["revision"]
            and len(keys) == 3 * N and not extra and not miss_key and not state_bad)
st_file_size = os.path.getsize(os.path.join(cd, "state.json"))
check("3c_state_json", state_ok,
      f"snapshot_revision={st['snapshot_revision']} (expected {last_p2_rev}); last_revision={st['last_revision']} "
      f"(expected {q['p3'][-1]['revision']}); keys={len(keys)} (expected {3 * N}); extra={len(extra)} missing={len(miss_key)} "
      f"per-key mismatches={len(state_bad)} {state_bad[:3]}; data_bytes={st['data_bytes']}; state.json on disk={st_file_size} bytes")

# --- 3d: paths.state_file ----------------------------------------------------
paths = summary.get("paths", {})
state_file_path = paths.get("state_file")
same = state_file_path == os.path.abspath(os.path.join(cd, "state.json")) and os.path.isfile(state_file_path) \
    and os.path.samefile(state_file_path, os.path.join(cd, "state.json"))
check("3d_state_file_path", bool(same), f"paths.state_file={state_file_path} same file as {cd}/state.json: {same}")

n_pass = sum(1 for _, ok, _ in results if ok)
print(f"\nSUMMARY: {n_pass}/{len(results)} checks PASS")
sys.exit(0 if n_pass == len(results) else 1)
